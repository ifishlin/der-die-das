def story_lines(n):
    txt = (HERE / "story_v4.md").read_text()
    sec = re.split(r"^## ", txt, flags=re.M)
    body = next(s for s in sec if s.startswith(f"段落 {n}　"))
    out = []
    for line in body.splitlines():
        if not line.startswith("> "):
            continue
        s = line[2:].strip()
        fs = [int(x) for x in re.findall(r"#(\d+)", s)]
        s = re.sub(r"\s*#\d+", "", s).strip()
        if s == "（停）" and out:
            out[-1]["text"] += "（停）"
            continue
        out.append({"text": s, "facts": fs})
    return out


segments = []
for n, name, src, action in SEGS:
    if src == "story":
        lines = story_lines(n)
    else:
        lines = [{"text": FIX.get(i, OLD[i]), "facts": FACTS.get(i, []), "from": i} for i in src]
    for k, l in enumerate(lines, 1):
        l["id"] = f"{n}-{k:02d}"
    segments.append({"n": n, "name": name, "action": action,
                     "lines": [{"id": l["id"], "text": l["text"], "facts": l["facts"]} for l in lines]})
(HERE / "script.json").write_text(json.dumps({"ep": "EP1-NEW", "segments": segments}, ensure_ascii=False, indent=1))
print(sum(len(s["lines"]) for s in segments), "句", len(segments), "段")
